# M3.10: VICTORY_INPLACE=1, windows of title memory placed in place: a heap
# opened on the title's private memory (by address), a buffer placed over
# it. Nothing is copied; shaders write title memory itself; while it stands
# the title cannot decommit it; a remap lets it go.
env VICTORY_INPLACE 1
env VICTORY_WINDOW_KB 256
region W watched 0x202400000000 0x800000
write W 0 0x800000 1
find a W 0x100010 0x100                  # placed (256 KiB at 0x100000)
find b W 0x100040 0x10                   # the same window
refresh a 0 0x40000                      # nothing to copy
write W 0x100100 0x10 2
refresh a 0 0x40000
prepare 1
gpu_declare a 0x100000 0x10000
card_write a 0x2000 0x10 3               # the card writes title memory itself
card_submit a 0x100000 0x10000
decommit W 0x100000 0x10000              # refused while in place: ok 0
find c W 0x100010 0x100                  # the remap let it go: placed anew
region P plain 0x202500000000 0x200000
find d P 0x10 0x100                      # plain private memory: placed
section S 0x400000
view V S 0 0x202600000000 0x400000
find e V 0x10 0x100                      # view memory: a copy
region R reserve 0x202700000000 0x400000
commit R 0 0x20000
find f R 0x10 0x100                      # partly committed: a copy
check
end
