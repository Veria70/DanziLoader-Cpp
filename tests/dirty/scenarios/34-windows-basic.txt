# Windows of title memory outside buffers (shadow.rs `lookup`, `loose`): an
# ask a title buffer holds gets the buffer's copy; any other gets a copy of
# its aligned 4 MiB window, on the card when `window_card_for` has room.
region W watched 0x201e00000000 0x1000000
write W 0 0x1000000 1
buffer b W 0x100000 0x100000
find x W 0x100010 0x100                 # inside b: b's copy
find y W 0x800010 0x100                 # a window, 4 MiB at 0x800000
find z W 0x900000 1                     # an ask without a length: the same window
find v W 0x3ffff0 0x20                  # across a 4 MiB line: an 8 MiB window
find u W 0x1fffff 0x2                   # past b's end: a window, not b's copy
refresh y 0 0x400000
refresh v 0 0x800000
write W 0x810000 0x40 2
write W 0x3ffff8 0x10 3
refresh y 0 0x400000
refresh v 0 0x800000
unbuffer b
find x2 W 0x100010 0x100                # b gone: now its window (0 to 4 MiB)
refresh x2 0 0x400000
prepare 1
gpu_declare y 0x10000 0x1000            # shaders declare a write into a window
card_write y 0x10010 0x10 4
card_submit y 0x10000 0x1000
refresh y 0 0x400000
end
