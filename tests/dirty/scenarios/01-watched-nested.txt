# Nested and overlapping host copies of one write-watched region share the
# journal: the kernel's dirty bit is cleared by whichever copy asks first.
region W watched 0x200000000000 0x100000
shadow outer W 0 0x100000
shadow inner W 0x23456 0x30000          # starts and ends mid-page
shadow tail W 0xf0001 0xffff            # the last page, from its second byte
refresh outer 0 0x100000
refresh inner 0 0x30000
refresh tail 0 0xffff
refresh outer 0 0x100000                # pages first read above count as written once
refresh inner 0 0x30000
refresh tail 0 0xffff
write W 0x30000 1 1                     # page 3
write W 0x5ffff 2 2                     # straddles pages 5 and 6
refresh inner 0 0x30000                 # inner asks first and clears the bits
refresh outer 0 0x100000                # outer learns of them from the journal
refresh tail 0 0xffff
write W 0x24000 0x100 3
refresh inner 0x1000 0x200              # partial-page request: the page stays stale
refresh inner 0 0x30000                 # the whole of it: taken
refresh outer -0x10000 0x20000          # a request that starts before the buffer
refresh outer 0xff000 0x10000           # and one that runs past its end
write W 0xf0000 0x10 4                  # the tail page, before its first byte
refresh tail 0 0xffff
refresh tail 0 0
refresh tail 0xffff 1
check
end
