# 11-env-ask-once-off.txt (switch variant of 06-ask-once.txt)
env VICTORY_ASK_ONCE 0
region A watched 0x200700000000 0x40000
region B watched 0x200700050000 0x40000  # 64 KiB after A: within the gap
region C watched 0x200800000000 0x20000  # far away
shadow a A 0 0x40000
shadow b B 0x100 0x3ff00
shadow c C 0 0x20000
flush a 0 0x40000 b 0 0x3ff00 c 0 0x20000
flush a 0 0x40000 b 0 0x3ff00 c 0 0x20000
write A 0x10 4 1
write B 0x20000 4 2
write C 0x10000 4 3
flush a 0 0x40000 b 0 0x3ff00 c 0 0x20000
ask_once a 0 0x40000 b 0 0x3ff00 c 0 0x20000
write A 0x30000 4 4
wrote A 0x30000 4
write C 0x0 4 5                          # not told: the standing asking misses it
refresh a 0 0x40000
refresh c 0 0x20000
ask_once_end
refresh c 0 0x20000
write B 0x0 1 6
ask_ahead a 0 0x40000 b 0 0x3ff00
refresh b 0 0x3ff00
ask_ahead_done
flush a 0 0x20000 a 0x20000 0x20000
check
end
