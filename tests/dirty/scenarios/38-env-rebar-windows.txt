# 38-env-rebar-windows.txt: VICTORY_REBAR=1, windows in CPU-visible card
# memory (64 KiB grid: the state lines read them over the bus).
env VICTORY_REBAR 1
env VICTORY_WINDOW_KB 64
region W watched 0x202200000000 0x400000
write W 0 0x400000 1
buffer b W 0x100000 0x100000
find x W 0x100010 0x100                 # a title buffer: not placed at level 1
find y W 0x200010 0x100                 # a window: placed
find z W 0x21fff0 0x20                  # across a line: 128 KiB, placed
refresh y 0 0x10000
refresh z 0 0x20000
write W 0x200100 0x10 2
write W 0x22fff0 0x20 3
refresh y 0 0x10000
refresh z 0 0x20000
end
