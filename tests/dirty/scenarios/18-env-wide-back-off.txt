# 18-env-wide-back-off.txt (switch variant of 14-copyback-basic.txt)
env VICTORY_WIDE_BACK 0
region W watched 0x200d00000000 0x80000
shadow g W 0x1000 0x60000
refresh g 0 0x60000
refresh g 0 0x60000
prepare 1
gpu_declare g 0x8000 0x10000
gpu_declare g 0x14000 0x8000            # overlaps the first
gpu_declare g 0x1c000 0x100             # touches the second
gpu_declare g 0x40001 0x1fff            # odd edges
snapshots
gpu_write g 0x8000 0x10 1               # a few bytes
gpu_write g 0x9000 0x800 2
gpu_write g 0x1bff0 0x20 3              # across the touching edge
gpu_write g 0x40001 0x1fff 4            # all of a range
write W 0x9008 0x10 5                   # the title writes under the shaders
write W 0x41004 4 6
hand_back g 0x8000 0x10000 g 0x14000 0x8000 g 0x1c000 0x100 g 0x40001 0x1fff
refresh g 0 0x60000                     # our writes show as title writes
check
prepare 2
gpu_declare g 0 0x60000                 # the whole copy
snapshots
gpu_write g 0x30000 0x30000 7
hand_back g 0 0x60000
refresh g 0 0x60000
end
