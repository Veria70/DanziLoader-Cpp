# Copy-back into guarded view memory: the write opens the guards (and every
# view of the pages), so a copy of another view sees it at its next refresh.
section S 0x40000
view A S 0 0x200f00000000 0x40000
view B S 0 0x201000000000 0x40000
shadow a A 0 0x40000
shadow b B 0x10000 0x20000
refresh a 0 0x40000
refresh b 0 0x20000
refresh a 0 0x40000
prepare 1
gpu_declare a 0x10000 0x18000
snapshots
gpu_write a 0x12000 0x100 1
gpu_write a 0x27ff0 0x10 2
write B 0x12010 0x8 3                    # the title through the other view
hand_back a 0x10000 0x18000
refresh b 0 0x20000                      # sees the GPU's bytes through its view
refresh a 0 0x40000
prepare 2
gpu_declare b 0 0x20000
snapshots
gpu_write b 0x4000 0x40 4
relay_back b 0 0x20000
gpu_done b 0 0x20000
refresh a 0 0x40000
check
end
