# 09-env-view-guard-off.txt (switch variant of 02-views-aliased.txt)
env VICTORY_VIEW_GUARD 0
section S 0x40000
view A S 0 0x200100000000 0x40000
view B S 0 0x200200000000 0x40000
view C S 0x20000 0x200300000000 0x20000   # the second half only
shadow a A 0 0x40000
shadow b B 0x8000 0x30000
shadow c C 0 0x20000
refresh a 0 0x40000
refresh b 0 0x30000
refresh c 0 0x20000
refresh a 0 0x40000
write B 0x30010 8 1                       # page 3 through B: page 3 of A, page 1 of C
refresh a 0 0x40000
refresh c 0 0x20000
refresh b 0 0x30000
write A 0 4 2                             # page 0 through A
refresh b 0 0x30000                       # b starts inside page 0
kwrite C 0x10000 0x100 3                  # a runtime write told at C
refresh a 0x30000 0x10000
refresh c 0 0x20000
refresh b 0 0x30000
write C 0x0 0x20000 4                     # all of C
flush a 0 0x40000 b 0 0x30000 c 0 0x20000
check
end
