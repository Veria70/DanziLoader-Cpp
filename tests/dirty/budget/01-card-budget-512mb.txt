# Milestone 3 (M3.8): a card with a small budget (512 MiB, set for the test
# in both runners). The reference cards all three 48 MiB copies (92% of the
# budget per copy, 1024 MiB in all). The port keeps carded copies, their
# hashes and its own buffers within 20% of the budget: it cards two, keeps
# the third in the PC's memory, and its ring stays in PC memory too. The
# results must not differ: compare with `dirty-diff --title` (title memory,
# title writes, the copies' bytes), not log for log.
env VICTORY_TEST_VRAM_BUDGET_MB 512
region W watched 0x201900000000 0x9000000
write W 0 0x9000000 1
card_shadow a W 0 0x3000000
card_shadow b W 0x3000000 0x3000000
card_shadow c W 0x6000000 0x3000000
refresh a 0 0x3000000
refresh b 0 0x3000000
refresh c 0 0x3000000
prepare 1
gpu_declare a 0x100000 0x200000
gpu_declare b 0 0x3000000
gpu_declare c 0x1000000 0x400000
snapshots
card_write a 0x100010 0x100 2
card_write b 0x2fff000 0x1000 3
card_write c 0x1000008 0x8 4
card_write c 0x13ffff0 0x10 5
card_submit a 0x100000 0x200000 b 0 0x3000000 c 0x1000000 0x400000
write W 0x6001000 0x20 6                 # the title, in c
write W 0x1000 0x20 7                    # and in a
refresh a 0 0x3000000
refresh b 0 0x3000000
refresh c 0 0x3000000
# Sent ahead, landed later.
prepare 2
gpu_declare c 0 0x3000000
gpu_declare a 0 0x3000000
snapshots
card_write c 0x2000000 0x10000 8
card_write a 0x2000000 0x10 9
card_send c 0 0x3000000 a 0 0x3000000
write W 0x6100000 0x10 10                # beside c's writes, while in the air
card_land
refresh a 0 0x3000000
refresh c 0 0x3000000
check
end
