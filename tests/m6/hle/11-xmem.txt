# M6.6: xmem: virtual memory with console flags, the system allocator,
# physical pages mapped into placeholders (and through DirectStorage-style
# copies), the working set and transfers. Addresses differ between runs:
# they are kept with `let` and only queried, never printed.
call xmem.dll XMemIsToolsMemAvailable:32
buf q 64
# Committed, 2 MiB pages: a region of graphics memory.
call xmem.dll XMemVirtualAlloc:0 null 0x400000 0x20003000 0 4
let a
call xmem.dll XMemVirtualQuery:32 -1 $a &q
dump q 40 24
call xmem.dll XMemVirtualQuery:32 -1 $a+0x10000 &q
dump q 40 24
# 32-bit and address-bit-limited memory; protection 0 is no access.
call xmem.dll XMemVirtualAlloc:0 null 0x10000 0x3000 4 4
let b
call xmem.dll XMemVirtualQuery:32 -1 $b &q
dump q 40 24
call xmem.dll XMemVirtualAlloc:0 null 0x10000 0x3000 0x3000 0
let c
call xmem.dll XMemVirtualQuery:32 -1 $c &q
dump q 40 24
# A reservation only: no flags reported.
call xmem.dll XMemVirtualAlloc:0 null 0x20000 0x2000 0 4
let r
call xmem.dll XMemVirtualQuery:32 -1 $r &q
dump q 40 24
call xmem.dll XMemVirtualQuery:32 -1 0 &q
call xmem.dll XMemVirtualQuery:32 -1 $r null
# 64 KiB pages: a placeholder, mapped with physical pages later.
call xmem.dll XMemVirtualAlloc:0 null 0x40000 0x20402000 0 4
let ph
call xmem.dll XMemVirtualQuery:32 -1 $ph &q
dump q 40 24
call xmem.dll XMemVirtualAlloc:0 $ph+0x10000 0x10000 0x20406000 0 4
call xmem.dll XMemVirtualQuery:32 -1 $ph &q
dump q 40 24
buf count 8
buf pages 64
store count 0 4
call xmem.dll XMemAllocatePhysicalPages:32 0 &count &pages
dump count 8
dump pages 32
call xmem.dll XMemAllocatePhysicalPages:32 0 null &pages
store count 0 0
call xmem.dll XMemAllocatePhysicalPages:32 0 &count &pages+32
dump count 8
store count 0 2
call xmem.dll XMemAllocatePhysicalPages:32 0 &count &pages+32
dump pages 16 32
# Mapped, then queried; the pages are written through a DirectStorage-like
# copy elsewhere (12-dstorage). Pages of two allocations: two views.
call xmem.dll XMemMapPhysicalPages:0 $ph 4 &pages
call xmem.dll XMemVirtualQuery:32 -1 $ph &q
dump q 40 24
call xmem.dll XMemVirtualQuery:32 -1 $ph+0x30000 &q
dump q 40 24
buf mixed 32
load p0 pages 0
load p5 pages 40
store mixed 0 $p0
store mixed 8 $p5
call xmem.dll XMemMapPhysicalPages:0 $ph+0x10000 2 &mixed
call xmem.dll XMemVirtualQuery:32 -1 $ph &q
dump q 40 24
call xmem.dll XMemVirtualQuery:32 -1 $ph+0x10000 &q
dump q 40 24
call xmem.dll XMemVirtualQuery:32 -1 $ph+0x20000 &q
dump q 40 24
# Unmapping part of a view keeps the rest; unknown pages fail.
call xmem.dll XMemMapPhysicalPages:0 $ph+0x10000 1 null
call xmem.dll XMemVirtualQuery:32 -1 $ph+0x10000 &q
dump q 40 24
call xmem.dll XMemVirtualQuery:32 -1 $ph+0x20000 &q
dump q 40 24
buf unknown 8
store unknown 0 0x12345
call xmem.dll XMemMapPhysicalPages:8 $ph 1 &unknown
# Scatter: one page an address, page 0 unmaps.
buf addrs 16
buf two 16
store addrs 0 $ph
store addrs 8 $ph+0x30000
load p1 pages 8
store two 0 $p1
store two 8 0
call xmem.dll XMemMapPhysicalPagesScatter:32 &addrs 2 &two
call xmem.dll XMemVirtualQuery:32 -1 $ph &q
dump q 40 24
call xmem.dll XMemVirtualQuery:32 -1 $ph+0x30000 &q
dump q 40 24
call xmem.dll XMemMapPhysicalPagesScatter:32 null 2 &two
call xmem.dll XMemMapPhysicalPagesScatter:32 &addrs 2 null
call xmem.dll XMemVirtualQuery:32 -1 $ph &q
dump q 40 24
call xmem.dll XMemFreePhysicalPages:32 4 &pages
call xmem.dll XMemFreePhysicalPages:32 4 null
call xmem.dll XMemMapPhysicalPages:8 $ph 1 &pages
# The allocator: no hooks without a title image.
call xmem.dll XMemAllocDefault:0 100 0
let m1
call xmem.dll XMemFreeDefault:0 $m1 0
call xmem.dll XMemAllocDefault:0 100 0xc000000
let m2
call xmem.dll XMemFreeDefault:0 $m2 0xc000000
call xmem.dll XMemAllocDefault:0 0x1000 0x200000000
let m3
call xmem.dll XMemVirtualQuery:32 -1 $m3 &q
dump q 40 24
call xmem.dll XMemFreeDefault:0 $m3 0x200000000
call xmem.dll XMemAllocDefault:8 -1 0
call xmem.dll XMemFreeDefault:0 null 0
call xmem.dll XMemAlloc:0 64 0
let m4
call xmem.dll XMemFree:0 $m4 0
# The budget: limits only (what the process uses differs).
buf ws 96
call xmem.dll XMemGetWorkingSetStatistics:32 0 &ws
dump ws 8
dump ws 8 16
dump ws 8 40
call xmem.dll XMemGetWorkingSetStatistics:32 0 null
call xmem.dll XMemTransferMemory:32 0 0
call xmem.dll XMemTransferMemory:32 0 3000000
call xmem.dll XMemTransferMemory:32 1 0x200000
call xmem.dll XMemTransferMemory:32 0 0xffffffffffffff00
call xmem.dll XMemTransferMemory:32 0 0x100000000
call xmem.dll XMemGetWorkingSetStatistics:32 0 &ws
dump ws 8
