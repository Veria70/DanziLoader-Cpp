# M4.7b: textures (time per call). Frames in small - reset, a texture copy,
# close, execute, a fence signalled behind (the relay, on by default) and
# waited for - reading back a tiled 64x64 texture the title leaves alone and
# one it writes each frame (detiled again), a tiled 1024x1024 one it writes
# each frame (4 MiB), and a buffer copied into a texture; then recording
# calls alone (texture copies, clears, a resolve).
runtime
create_device dev device11
create_queue dev q 0
create_fence dev f 0
allocator dev a 0
list dev l 0 a
list_close l
desc_heap dev rtvs 2 8
desc_heap dev dsvs 3 8
committed dev back 3 B4194304
committed dev up 2 B4194304
fill up 65536 3 7
committed dev til 1 T3/64x64/f28
fill til 16384 13 1
committed dev big 1 T3/1024x1024/f28
fill big 4194304 5 3
committed dev dt 1 T3/32x32/f28
committed dev rt 1 T3/256x256/f28/F1
rtv dev rtvs:0 rt
committed dev dp 1 T3/256x256/f40/F2
dsv dev dsvs:0 dp
committed dev ms 1 T3/64x64/f28/F1/s4
committed dev rs 1 T3/64x64/f28
list dev lr 0 a
quiet 1
repeat 2000 texture_cycle q l a f back:fp:0:28:64:64:1:256 til:sub:0 til 0
repeat 2000 texture_cycle q l a f back:fp:0:28:64:64:1:256 til:sub:0 til 16384
repeat 200 texture_cycle q l a f back:fp:0:28:1024:1024:1:4096 big:sub:0 big 4194304
repeat 2000 texture_cycle q l a f dt:sub:0 up:fp:0:28:32:32:1:128 up 0
repeat 10000 copy_tex lr back:fp:0:28:64:64:1:256 0 0 0 til:sub:0
repeat 10000 copy_tex lr dt:sub:0 0 0 0 up:fp:0:28:32:32:1:128
repeat 10000 clear_rtv lr rtvs:0 1,0.5,0.25,1
repeat 10000 clear_dsv lr dsvs:0 1 0.75 0
repeat 10000 resolve lr rs 0 ms 0 28
repeat 2000 texture_cycle q l a f back:fp:0:28:64:64:1:256 til:sub:0 til 0
repeat 2000 texture_cycle q l a f back:fp:0:28:64:64:1:256 til:sub:0 til 16384
repeat 200 texture_cycle q l a f back:fp:0:28:1024:1024:1:4096 big:sub:0 big 4194304
repeat 2000 texture_cycle q l a f dt:sub:0 up:fp:0:28:32:32:1:128 up 0
repeat 10000 copy_tex lr back:fp:0:28:64:64:1:256 0 0 0 til:sub:0
repeat 10000 copy_tex lr dt:sub:0 0 0 0 up:fp:0:28:32:32:1:128
repeat 10000 clear_rtv lr rtvs:0 1,0.5,0.25,1
repeat 10000 clear_dsv lr dsvs:0 1 0.75 0
repeat 10000 resolve lr rs 0 ms 0 28
end
